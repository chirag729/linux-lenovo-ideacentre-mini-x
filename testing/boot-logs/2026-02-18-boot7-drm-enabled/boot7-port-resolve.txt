
echo "---RESOLVE-PHANDLES---"
for p in 0 4 5 6; do
  ep="/sys/firmware/devicetree/base/soc@0/display-subsystem@ae00000/display-controller@ae01000/ports/port@$p/endpoint/remote-endpoint"
  ph=$(od -A n -t x1 "$ep" 2>/dev/null | tr -d ' \n')
  echo "port@$p remote-phandle: 0x$ph"
done
echo "---FIND-NODE-BY-PHANDLE---"
find /sys/firmware/devicetree/base -name phandle -exec sh -c '
  ph=$(od -A n -t x1 "$1" 2>/dev/null | tr -d " \n")
  case "$ph" in
    000000f1|000000ea|000000e5|000000ec|000000e0|000000e8)
      echo "phandle 0x$ph -> $(dirname "$1")";;
  esac
' _ {} \;
